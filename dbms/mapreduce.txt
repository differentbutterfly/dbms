// =========================================
// MAPREDUCE BASICS
// map() → emits key-value pairs
// reduce() → combines values for each key
// =========================================

// STEP 3: Total price per customer
var map = function () {
  emit(this.Cust_id, this.Amt);
};

var reduce = function (key, values) {
  return Array.sum(values);
};

db.orders.mapReduce(map, reduce, { out: "neworders" });
db.neworders.find().pretty();

// Expected output:
// { "_id": "A1", "value": 95000 }
// { "_id": "B1", "value": 52000 }
// { "_id": "C1", "value": 36000 }

// STEP 4: Total price per customer where Status = 'D'
db.orders.mapReduce(map, reduce, {
  query: { Status: "D" },
  out: "new1"
});
db.new1.find().pretty();

// Expected output:
// { "_id": "A1", "value": 90000 }
// { "_id": "B1", "value": 12000 }

// STEP 5: Total price per customer where Status = 'P'
db.orders.mapReduce(map, reduce, {
  query: { Status: "P" },
  out: "new2"
});
db.new2.find().pretty();

// Expected output:
// { "_id": "A1", "value": 5000 }
// { "_id": "B1", "value": 40000 }
// { "_id": "C1", "value": 36000 }

// =========================================
// ADVANCED MAPREDUCE
// Find count of all keys (field names) in the orders collection
// =========================================

var map1 = function () {
  for (var key in this) {
    emit(key, { CountOfKey: 1 });
  }
};

var reduce1 = function (key, values) {
  var total = 0;
  for (var i = 0; i < values.length; i++) {
    total += values[i].CountOfKey;
  }
  return { CountOfKey: total };
};

db.orders.mapReduce(map1, reduce1, { out: "new3" });
db.new3.find().pretty();

// Expected output example:
// { "_id": "Amt", "value": { "CountOfKey": 6 } }
