-- University Database Creation
CREATE DATABASE University;
USE University;

-- 1. Department table with Unique constraint
CREATE TABLE Department(
    dept_no INT,
    dept_name VARCHAR(15),
    bldg_name VARCHAR(15),
    UNIQUE(dept_name)
);

-- 2. Instructor table with NOT NULL constraint
CREATE TABLE instructor(
    ins_id INT,
    ins_name VARCHAR(15) NOT NULL,
    dept_no INT,
    salary INT,
    mob_no INT,
    PRIMARY KEY(ins_id)
);

-- 3. Course table
CREATE TABLE course(
    course_id INT,
    title VARCHAR(15),
    dept_no INT,
    credits INT,
    PRIMARY KEY(course_id)
);

-- 4. Teaches table with Foreign Keys
CREATE TABLE teachers(
    teacher_id INT,
    course_id INT,
    semester INT,
    year YEAR,
    FOREIGN KEY (teacher_id) REFERENCES instructor(ins_id),
    FOREIGN KEY (course_id) REFERENCES course(course_id)
);
