  _id: 1,
  Title: "DBMS",
  Content: "Structured Database",
  Author: "Silber",
  Author_age: 50,
  DOP: ISODate("2002-06-07"),
  Category: "Comp",

  ]
});

// STEP 11: Add one more comment to specific title
db.ar.update(
  { Title: "TOC" },
  { $push: { Comment: { Name: "Hope", Remarks: "Bad" } } }
);

// STEP 12: Change category from IT → Information Technology
db.ar.updateMany(
  { Category: "IT" },
  { $set: { Category: "Information Technology" } }
);

// STEP 13: Change DOP for author “Silber”
db.ar.update(
  { Author: "Silber" },
  { $set: { DOP: ISODate("2001-03-24") } }
);

// STEP 14: Update document if found else insert (upsert)
db.ar.update(
  { _id: 1 },
  { $set: { Content: "Database" } },
  { upsert: true }
);

// Insert new doc with upsert (not found case)
db.ar.update(
  { Title: "ADS" },
  {
    $set: {
      Title: "ADS",
      Content: "Data Structure",
      Author: "Horowitz",
      Author_age: 50,
      DOP: ISODate("2010-04-23")
    }
  },
  { upsert: true }
);

// STEP 15: Find authors whose age > 40
db.ar.find({ Author_age: { $gt: 40 } }).pretty();

// STEP 16: Find authors with age between 50 and 80
db.ar.find({ Author_age: { $gt: 50, $lt: 80 } }).pretty();

// STEP 17: Insert multiple comments for a title
db.ar.update(
  { _id: 1 },
  {
    $push: {
      Comment: {
        $each: [
          { Name: "Kyle", Remarks: "Good book" },
          { Name: "Danny", Remarks: "Okay" }
        ]
      }
    }
  }
);

// STEP 18: Display docs with Title=Java OR Category=Comp
db.ar.find({ $or: [{ Title: "Java" }, { Category: "Comp" }] }).pretty();

// STEP 19: Delete first article by a specific author
db.ar.remove({ Author: "Forouzan" }, { justOne: true });

// STEP 20: Delete all articles with Category = Comp
db.ar.remove({ Category: "Comp" }, { multi: true });
